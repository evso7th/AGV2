🛠️ 3. Рефакторинг buildGuitarEngine (instrument-factory (14).ts)
3.1. Ядро: Гибридная модель Karplus-Strong + Транзиент
Заменяем статический PeriodicWave на физически-корректную цепочку. CPU-лёгкая, полифония сохраняется через ARP.
/** #ОБНОВЛЕНО (ПЛАН №23010): RealString Hybrid Engine */
const buildGuitarEngine = (ctx: AudioContext, preset: any, master: GainNode, reverb: ConvolverNode, instrumentGain: GainNode, expressionGain: GainNode) => {
  let currentPreset = { ...preset };
  
  // Цепочка эффектов (сохранена для совместимости)
  const comp = ctx.createDynamicsCompressor(); comp.threshold.value = -18;
  const shaper = ctx.createWaveShaper(); shaper.oversample = '4x';
  const satNode = ctx.createWaveShaper(); satNode.curve = null;
  const phaser = makePhaser(ctx, currentPreset.phaser || {});
  const delay = makeDelay(ctx, currentPreset.delayA || currentPreset.delay || {});
  const revSend = ctx.createGain(); revSend.gain.value = isFinite(currentPreset.reverbMix) ? currentPreset.reverbMix : 0.2;
  const cabinetFilter = ctx.createBiquadFilter(); cabinetFilter.type = 'lowpass';
  
  const activeVoiceRecords = new Set<any>();
  const bodyResonance = ctx.createBiquadFilter(); bodyResonance.type = 'peaking';
  
  // Инициализация фильтров
  const updateCabinet = (p: any) => {
    cabinetFilter.frequency.value = p.post?.lpf ?? 4000;
    bodyResonance.frequency.value = p.string?.bodyResFreq ?? 160;
    bodyResonance.gain.value = p.string?.bodyResGain ?? 0.15;
    bodyResonance.Q.value = 1.8;
  };
  updateCabinet(currentPreset);

  // Цепь: Вход -> Комп -> Дист -> Фазер -> Дилей -> Корпус -> Cabinet -> Выход
  const preGain = ctx.createGain(); preGain.gain.value = 1.0;
  preGain.connect(comp).connect(shaper).connect(satNode).connect(phaser.input);
  phaser.output.connect(delay.input);
  delay.output.connect(cabinetFilter).connect(bodyResonance).connect(expressionGain).connect(instrumentGain).connect(master);
  cabinetFilter.connect(revSend).connect(reverb);

  // 🆕 Математический транзиент медиатора
  const createPickTransient = (velocity: number) => {
    const len = Math.floor(ctx.sampleRate * 0.012);
    const buf = ctx.createBuffer(1, len, ctx.sampleRate);
    const d = buf.getChannelData(0);
    const decay = ctx.sampleRate * 0.002;
    for (let i = 0; i < len; i++) d[i] = (Math.random() * 2 - 1) * Math.exp(-i / decay) * velocity;
    return buf;
  };

  // 🆕 Струнный голос (Karplus-Strong + Harmonic Decay)
  const createStringVoice = (freq: number, velocity: number, duration: number) => {
    const p = currentPreset;
    const damp = clamp(p.string?.damping ?? 0.65, 0.1, 0.95);
    const bright = clamp(p.string?.brightness ?? 0.7, 0.2, 0.95);
    
    // Задержка = период основной частоты
    const delayNode = ctx.createDelay(2.0); delayNode.delayTime.value = 1 / freq;
    
    // Усреднитель + Нелинейное затухание
    const avg = ctx.createGain(); avg.gain.value = 0.5;
    const dampFilter = ctx.createBiquadFilter(); dampFilter.type = 'lowpass';
    dampFilter.frequency.value = freq * (4 + (1 - bright) * 8);
    dampFilter.Q.value = 0.7;
    
    // Цепочка обратной связи
    delayNode.connect(avg).connect(dampFilter).connect(delayNode);
    
    // Возбуждение (шумовой импульс + транзиент)
    const exciteLen = Math.floor(ctx.sampleRate * 0.02);
    const exciteBuf = ctx.createBuffer(1, exciteLen, ctx.sampleRate);
    const exciteData = exciteBuf.getChannelData(0);
    for (let i = 0; i < exciteLen; i++) exciteData[i] = (Math.random() * 2 - 1) * Math.exp(-i / (ctx.sampleRate * 0.003)) * velocity;
    
    const exciteSrc = ctx.createBufferSource(); exciteSrc.buffer = exciteBuf;
    const pickSrc = ctx.createBufferSource(); pickSrc.buffer = createPickTransient(velocity);
    const exciteGain = ctx.createGain(); exciteGain.gain.value = 1.2;
    const pickGain = ctx.createGain(); pickGain.gain.value = p.noise?.pick?.gain ?? 0.3;
    
    exciteSrc.connect(exciteGain).connect(delayNode);
    pickSrc.connect(pickGain).connect(exciteGain); // Транзиент вливается в струну
    
    exciteSrc.start(); pickSrc.start();
    
    // ADSR + Release для K-S (экспоненциальный хвост)
    const outGain = ctx.createGain(); outGain.gain.value = 0;
    delayNode.connect(outGain);
    
    const now = Math.max(ctx.currentTime, 0);
    const adsr = getADSR(p);
    const atk = Math.max(adsr.a, 0.002);
    const sust = clamp(adsr.s * velocity, 0.05, 1.0);
    outGain.gain.setValueAtTime(0.0001, now);
    outGain.gain.linearRampToValueAtTime(velocity, now + atk);
    outGain.gain.setTargetAtTime(sust, now + atk, adsr.d / 3);
    
    const relTime = now + duration;
    const rel = Math.max(adsr.r, 0.1);
    outGain.gain.setTargetAtTime(0.0001, relTime, rel / 2.5);
    
    const finalStop = relTime + (rel * 5);
    exciteSrc.stop(finalStop); pickSrc.stop(finalStop);
    
    return { nodes: [delayNode, avg, dampFilter, exciteSrc, pickSrc, exciteGain, pickGain, outGain], endTime: finalStop, output: outGain };
  };

  return {
    noteOn: (midi: number, when = ctx.currentTime, velocity = 1.0, duration = 1.0, technique?: string) => {
      if (!isFinite(midi)) return;
      reapVoices(ctx.currentTime);
      const f = midiToHz(midi);
      
      // 🆕 Velocity → Spectral Mapping
      const velNorm = clamp(velocity, 0.1, 1.0);
      const brightMod = (currentPreset.velocityMap?.brightnessMod ?? 0.4) * velNorm;
      const driveMod = (currentPreset.velocityMap?.driveMod ?? 0.3) * velNorm;
      
      // Динамический дисторшн + кабинет
      const driveAmt = (currentPreset.drive?.amount ?? 0) + driveMod;
      shaper.curve = driveAmt < 0.05 ? null : (currentPreset.drive?.type === 'muff' ? makeMuff(driveAmt) : makeVintageDistortion(driveAmt * 50));
      cabinetFilter.frequency.setTargetAtTime((currentPreset.post?.lpf ?? 4000) * (0.7 + 0.3 * velNorm), ctx.currentTime, 0.01);

      // Создание голоса
      const voice = createStringVoice(f, velNorm, duration);
      voice.output.connect(preGain);
      
      const record = { nodes: voice.nodes, voiceState: { node: voice.output, startTime: when }, cleaned: false, endTime: voice.endTime };
      globalActiveVoices.push(record);
      activeVoiceRecords.add(record);

      // 🆕 Техники (TECHNIQUE_GUIDE.md)
      if (technique) applyTechnique(when, duration, voice, technique, currentPreset);
    },
    allNotesOff: () => { activeVoiceRecords.forEach(cleanupNow); activeVoiceRecords.clear(); },
    disconnect: () => { 
      phaser.disconnect(); delay.disconnect(); activeVoiceRecords.forEach(cleanupNow); 
      [preGain, comp, shaper, satNode, revSend, cabinetFilter, bodyResonance].forEach(n => { try{n.disconnect();}catch(e){} }); 
    },
    setPreset: (p: any) => { 
      currentPreset = { ...p }; updateCabinet(p); 
      shaper.curve = p.drive?.amount < 0.05 ? null : (p.drive?.type === 'muff' ? makeMuff(p.drive.amount) : makeVintageDistortion(p.drive.amount * 50));
    }
  };
};
3.2. Контроллер техник (TECHNIQUE_GUIDE.md → Synth Params)
Добавляется в тот же файл или в отдельный guitar-techniques.ts. Вызывается внутри noteOn.
const applyTechnique = (when: number, duration: number, voice: any, tech: string, preset: any) => {
  const { nodes, output } = voice;
  const mainGain = nodes.find((n: any) => n instanceof GainNode && n !== nodes[5] && n !== nodes[6]) as GainNode;
  if (!mainGain) return;

  const t = preset.technique || {};
  switch (tech) {
    case 'sl': // Slide
      mainGain.gain.setTargetAtTime(0, when + 0.001, 0.005); // Micro gap
      break;
    case 'bn': // Bend
      // Pitch ramp через detune всех осцилляторов/delay
      nodes.filter((n: any) => n.delayTime).forEach((d: any) => {
        const bendCents = t.bend?.maxCents ?? 150;
        d.delayTime.setTargetAtTime(d.delayTime.value * Math.pow(2, -bendCents/1200), when + 0.01, t.bend?.speed ?? 0.02);
      });
      break;
    case 'vb': // Vibrato
      const vibLfo = new (ctx as any).Oscillator(); // WebAudio LFO
      vibLfo.frequency.value = t.vibrato?.rate ?? 5.5;
      const vibG = ctx.createGain(); vibG.gain.value = t.vibrato?.depth ?? 6;
      vibLfo.connect(vibG);
      nodes.filter((n: any) => n.delayTime).forEach(d => vibG.connect(d.delayTime));
      vibLfo.start(when); vibLfo.stop(when + duration);
      break;
    case 'F_TRAVIS': // Staggered attack (polyphony)
      // Реализуется на уровне Scheduler, здесь микро-сдвиг envelope
      mainGain.gain.linearRampToValueAtTime(0.8, when + 0.012);
      break;
    default: break;
  }
};

⚠️ 6. Ограничения и рекомендации
iOS Safari Background: MediaStream-хак (BroadcastEngine) обязателен. Web Audio таймеры в фоне дрейфуют на ±30мс → Scheduler должен компенсировать джиттер через AudioContext.currentTime.
CPU Budget: Karplus-Strong × 120 голосов ≈ 2-3% CPU на современном мобильном чипе. При троттлинге снижать currentVoiceLimit до 60 через setGlobalVoiceLimit().
Humanization Layer: Движок даёт физическую аутентичность, но для 90% в слепом тесте Worker должен добавлять ±5-12ms микро-сдвиги старта нот, ±1.5 cents дрейф питча и амплитудную флуктуацию ±3%. Без этого звук будет "идеальным", но "роботизированным".
DNA/Brain Integration: Аксиомы из heritage_axioms должны передавать technique и velocity в noteOn. Если technique отсутствует, движок выбирает дефолт из пресета.
